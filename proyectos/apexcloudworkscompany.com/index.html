import Anthropic from "@anthropic-ai/sdk";
import AWS from "aws-sdk";
const client = new Anthropic();
const s3 = new AWS.S3();
// Genera landing desde el brief del cliente
async function generateLanding(brief) {
  const res = await client.messages.create({
    model: "claude-sonnet-4-6",
    max_tokens: 8192,
    messages: [{ role: "user",
      content: `Diseñá landing para: ${brief}`}]
  });
  return res.content[0].text;
}
// Deploy a AWS S3 + CloudFront
const deploy = async (html, bucket) => {
  await s3.putObject({ Bucket: bucket,
    Key: "index.html", Body: html }).promise();
  console.log("✓ Live en producción");
};
